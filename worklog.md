- Key architectural patterns to mirror:
  - Pure-function engine: `engine.sample(t)` is a pure function of time. No
    Date.now(), no internal mutable state. Enables pause/resume/seek/seek-back
    determinism, screenshot export at any frame, and DOM-less unit tests.
  - Transitions are EXPONENTIAL EASE-OUTS, NOT springs. Body NEVER overshoots.
    easeOutCubic / easeInOutCubic / easeOutQuint. The only spring-like effect
    in the entire project is NOTIF_POP = 1.14 (notification pastille pop).
  - Silhouettes as radial profiles sampled at PROFILE_SAMPLES=64 angles. Any
    shape transition is a linear interpolation of radii arrays. No
    path-morphing library needed.
  - Eyes are mask HOLES, not white shapes on top. Two white ellipses morph
    independently. Body is backed by opaque paper-color path so the holes
    don't show particles/rings behind.
  - Eyes live on a sphere (orthographic projection). yaw/pitch/roll drive a
    3D rotation; near-edge eyes self-clip via the mask. radiusAtAngle keeps
    eyes on the real outline for non-circular bodies.
  - Eye placement for custom (non-circular) bodies is solved ONCE at import
    via a pose-space-deformation lookup table (eyefit.ts) — NOT per-frame.
    Per-frame solvers chattered and switched active-set; the table is pure
    data and deterministic. (Pose Space Deformation, Lewis/Cordner/Fong
    SIGGRAPH 2000.)
  - Look target is ABSOLUTE yaw/pitch, mixed into the pose by the engine
    (only it knows the current pose during a morph). mix = how much the
    outside world commands direction; wander = residual automatic drift,
    ADDED AFTER the mix so the head keeps living when no pointer is present.
  - setState during a fade FREEZES the composite pose and blends from it
    (continuity across chained state changes).
  - Single rAF loop, bounded dt (Math.min((ms-last)/1000, 0.064)) to survive
    tab-unhide. Single render path: sample(clock) → triggerRef. No substeps.
  - Pointer perception: window 'pointermove' + document 'pointerleave', no
    Rive listener. Re-read getBoundingClientRect each frame (avatar slides
    and grows during transitions). Normalize against half-WINDOW, not avatar
    size, so gaze saturates at screen edge. Reject touch input.
  - Particle overlay: SAME SVG canvas as the body, just a separate <g>
    layer behind the body (dotsBehind: true). Spirals inward
    (rho *= 0.75^(u*10), angle += 100deg/s), grows, fades, depth-fogs via
    mixHex(paper, ink, depth).
  - Three views: Customise / Animations / Settings. Exports SVG/PNG/GIF/MP4.

  // math.ts — the entire easing catalog (note: no spring integrator)
  export const easings = {
    easeOutCubic: (t: number) => 1 - (1 - t) ** 3,
    easeInOutCubic: (t: number) =>
      (t < 0.5 ? 4 * t ** 3 : 1 - (-2 * t + 2) ** 3 / 2),
    easeOutQuint: (t: number) => 1 - (1 - t) ** 5
  } satisfies Record<string, Easing>
  // math.ts comment (translated): "Measured on the video: transitions are
  // exponential ease-outs, without body overshoot. The only spring effects
  // are local (notification pastille pop, eye opening) and are written
  // directly in the relevant state."

  // engine.ts — morph durations (seconds)
  static readonly SHAPE_MORPH = 0.45
  static readonly LOOK_MORPH = 0.24 // "shorter than SHAPE_MORPH: a tracking
                                    //  gaze must look attentive, not viscous"

  // decor.ts — the only spring-like constant
  export const NOTIF_POP = 1.14 // pop peaks 14% above rest at ~0.3s, then settles

  // BloubBot.vue — single rAF loop, bounded dt
  function tick(ms: number) {
    raf = requestAnimationFrame(tick)
    const dt = last ? Math.min((ms - last) / 1000, 0.064) : 0
    last = ms
    clock += dt
    if (playing.value) {
      if (clock >= nextAt && props.cycle.length)
        goToBlock((block.value + 1) % props.cycle.length)
      else
        elapsed.value = clock - blockStart
    }
    if (props.follow) aim()
    else if (props.gaze) scriptedGaze(props.gaze)
    frame.value = engine.sample(clock)
    triggerRef(frame)
  }

  // BloubBot.vue — pointer perception (no Rive listener)
  function onPointerMove(event: PointerEvent) {
    if (event.pointerType === 'touch') return  // touch would freeze gaze
    pointer = { x: event.clientX, y: event.clientY }
  }
  function aim() {
    if (!STATE_BY_ID.get(state.value)?.baseFace) { release(); return }
    const box = svg.value?.getBoundingClientRect()
    if (!box || box.width === 0 || box.height === 0) return  // NaN guard
    if (!aiming) turnSince = clock
    const demiLargeur = Math.max(1, window.innerWidth / 2)
    const demiHauteur = Math.max(1, window.innerHeight / 2)
    engine.setLook(
      lookTarget({
        nx: pointer ? clamp((pointer.x - (box.left + box.width/2)) / demiLargeur, -1, 1) : 0,
        ny: pointer ? clamp((pointer.y - (box.top + box.height/2)) / demiHauteur, -1, 1) : 0,
        tour: easings.easeOutQuint(clamp((clock - turnSince) / TURN_TIME)),
        pointer: pointer !== null
      }),
      clock
    )
    aiming = true
  }

  // gaze.ts — pointer → Look mapping (absolute yaw/pitch, mix, spin, wander)
  export const YAW_MAX = 16; export const PITCH_MAX = 13; export const PITCH = 10
  export const TURN = 26; export const SPIN = 360; export const TURN_TIME = 1.1
  export function lookTarget({ nx, ny, tour, pointer }: Aim): Look {
    return {
      yaw: -TURN + nx * YAW_MAX,
      pitch: PITCH - ny * PITCH_MAX,
      mix: tour,
      spin: SPIN * (1 - tour),
      wander: pointer ? 0 : 1
    }
  }
